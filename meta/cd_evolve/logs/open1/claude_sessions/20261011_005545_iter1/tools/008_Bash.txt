"""Stage 0 for Multi-Turn Puzzles CircuitDecoding: exact hypothesis space,
oracle solver, and the per-query information-efficiency measure.

No model calls. Pure CPU.

Why this family and not GuessNumbers: in GN almost every consistent guess is
near-maximally informative (oracle 5.22 turns vs random-consistent 5.35), so
there is no "legal but uninformative" move for a belief trap to consist of.
Here feedback is a single bit, and querying an entry the history already
determines is perfectly legal, non-contradictory, and worth exactly 0 bits.
That is the move a trapped agent makes.

Structure, verified against all 300 instances of arianhosseini/mt_puzzles:
  C = 3 circuits, k = 3 inputs, 24-bit joint truth table.
  Each circuit is built by reducing k signals to 1, so it uses exactly k-1 = 2
  binary gates (AND/OR, each consumes two signals and yields one); NOT replaces
  one signal in place and so is unconstrained in count. Binary gates always
  total 6 = 3 x 2. The component inventory is told to the agent and is the
  prior that makes 18 queries enough for 24 bits.
  Only three inventories occur, 100 instances each:
      {AND 1, OR 5, NOT 2} / {AND 3, OR 3, NOT 0} / {AND 3, OR 3, NOT 1}
  Truth-table row order is 000..111 with x[0] as the most significant input;
  stored as a string, first row leftmost. Checked on all 900 circuits.
"""

import json
from functools import lru_cache
from itertools import permutations, product
from math import log2
from pathlib import Path

import numpy as np

K = 3
NROWS = 1 << K          # 8 rows per circuit
FULL = (1 << NROWS) - 1  # all-ones truth table
# x[0] is the MSB of the row index: x0 = 00001111, x1 = 00110011, x2 = 01010101
INPUTS = tuple(
    int("".join(str((r >> (K - 1 - i)) & 1) for r in range(NROWS)), 2)
    for i in range(K)
)


def tt_to_str(tt):
    return format(tt, f"0{NROWS}b")


def str_to_tt(s):
    return int(s, 2)


# ---------------------------------------------------------------- enumeration


@lru_cache(maxsize=None)
def realizable(a, o, n):
    """Truth tables a single circuit can realise using exactly a ANDs, o ORs
    and n NOTs, starting from the k inputs and ending on one signal.
    Constant circuits are excluded (the task guarantees none)."""
    out = set()

    def walk(signals, a, o, n):
        if len(signals) == 1 and a == 0 and o == 0 and n == 0:
            if signals[0] not in (0, FULL):
                out.add(signals[0])
            return
        sig = list(signals)
        if n:
            for i in range(len(sig)):
                nxt = sig[:i] + [FULL ^ sig[i]] + sig[i + 1:]
                walk(tuple(sorted(nxt)), a, o, n - 1)
        if a or o:
            for i in range(len(sig)):
                for j in range(i + 1, len(sig)):
                    rest = [sig[x] for x in range(len(sig)) if x not in (i, j)]
                    if a:
                        walk(tuple(sorted(rest + [sig[i] & sig[j]])), a - 1, o, n)
                    if o:
                        walk(tuple(sorted(rest + [sig[i] | sig[j]])), a, o - 1, n)

    walk(tuple(sorted(INPUTS)), a, o, n)
    return frozenset(out)


def binary_splits(n_and, n_or, c=3):
    """Ways to hand each labelled circuit exactly two binary gates."""
    seen = set()
    for perm in set(permutations(["A"] * n_and + ["O"] * n_or)):
        key = tuple(
            (perm[2 * i: 2 * i + 2].count("A"), perm[2 * i: 2 * i + 2].count("O"))
            for i in range(c)
        )
        seen.add(key)
    return sorted(seen)


def not_splits(n_not, c=3):
    return [t for t in product(range(n_not + 1), repeat=c) if sum(t) == n_not]


def hypothesis_space(components, c=3):
    """All joint truth tables consistent with the inventory, as an (H, 24)
    uint8 array of bits (circuit-major, row order 000..111)."""
    a, o, n = (components.get("AND", 0), components.get("OR", 0),
               components.get("NOT", 0))
    joint = set()
    for bsplit in binary_splits(a, o, c):
        for nsplit in not_splits(n, c):
            per = [realizable(bsplit[i][0], bsplit[i][1], nsplit[i])
                   for i in range(c)]
            if not all(per):
                continue
            joint.update(product(*per))
    arr = np.zeros((len(joint), c * NROWS), dtype=np.uint8)
    for h, tup in enumerate(sorted(joint)):
        for ci, tt in enumerate(tup):
            s = tt_to_str(tt)
            arr[h, ci * NROWS:(ci + 1) * NROWS] = [int(ch) for ch in s]
    return arr


# ---------------------------------------------------------------- the game


class CDGame:
    """Exact belief bookkeeping over CircuitDecoding.

    A query is an index 0..23 into the joint truth table: circuit q // 8,
    input row q % 8. The answer is that bit. Everything below is computed on
    the feasible set, never on the agent's internal belief."""

    N_QUERIES = 24

    def __init__(self, components):
        self.components = dict(components)
        self.H = hypothesis_space(components)

    def feasible(self, history):
        alive = np.arange(len(self.H))
        for q, bit in history:
            alive = alive[self.H[alive, q] == bit]
        return alive

    def expected_bits(self, q, alive):
        """Entropy of the answer to query q over the feasible set. At most 1
        bit; exactly 0 when the history already pins that entry."""
        if len(alive) <= 1:
            return 0.0
        p = self.H[alive, q].mean()
        if p in (0.0, 1.0):
            return 0.0
        return float(-(p * log2(p) + (1 - p) * log2(1 - p)))

    def available_bits(self, alive):
        """Best any legal query can do right now -> (bits, query index)."""
        if len(alive) <= 1:
            return 0.0, None
        p = self.H[alive].mean(0)
        with np.errstate(divide="ignore", invalid="ignore"):
            ent = np.where((p > 0) & (p < 1),
                           -(p * np.log2(p) + (1 - p) * np.log2(1 - p)), 0.0)
        q = int(ent.argmax())
        return float(ent[q]), q

    def efficiency(self, alive, q):
        """Ex-ante efficiency in [0, 1]: what the chosen query is worth over
        what the best query is worth. Free of outcome luck."""
        avail, _ = self.available_bits(alive)
        chosen = self.expected_bits(q, alive)
        return (float("nan") if avail <= 0 else chosen / avail), chosen, avail

    # ---- policies ----------------------------------------------------

    def solve_greedy(self, truth, budget=18):
        """Max-entropy oracle. Returns the number of queries needed to pin the
        joint truth table uniquely, or None if the budget runs out."""
        alive = np.arange(len(self.H))
        for t in range(1, budget + 1):
            if len(alive) == 1:
                return t - 1
            _, q = self.available_bits(alive)
            alive = alive[self.H[alive, q] == truth[q]]
        return None if len(alive) > 1 else budget

    def solve_random(self, truth, rng, budget=18):
        """Null policy: query a uniformly random entry, repeats allowed --
        exactly the behaviour that is legal but can waste a turn."""
        alive = np.arange(len(self.H))
        for t in range(1, budget + 1):
            if len(alive) == 1:
                return t - 1
            q = int(rng.integers(self.N_QUERIES))
            alive = alive[self.H[alive, q] == truth[q]]
        return None if len(alive) > 1 else budget

    def solve_sequential(self, truth, budget=18):
        """Null policy 2: sweep entries 0,1,2,... in order. Legal, never
        repeats, but ignores what the inventory already implies."""
        alive = np.arange(len(self.H))
        for t in range(1, budget + 1):
            if len(alive) == 1:
                return t - 1
            q = (t - 1) % self.N_QUERIES
            alive = alive[self.H[alive, q] == truth[q]]
        return None if len(alive) > 1 else budget


def load_instances(path):
    out = []
    for line in open(path):
        c = json.loads(line)["controller"]
        out.append({
            "components": c["components"],
            "truth": np.array([int(ch) for ch in c["circuit_joint_lookup"]],
                              dtype=np.uint8),
            "circuits": c["circuits"],
        })
    return out
