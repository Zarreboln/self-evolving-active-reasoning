import json, sys, collections
from pathlib import Path
import numpy as np

def load(path):
    latest = {}
    for line in Path(path).open():
        if line.strip():
            r = json.loads(line)
            latest[(r["key"], r["seed"])] = r
    return latest

for name in sys.argv[1:]:
    eps = load(f"logs/open1/traces/{name}/search.jsonl")
    n_scan = n_q = 0
    n_scan_from_A0 = 0
    zero_by_alive = collections.Counter()
    q_by_alive = collections.Counter()
    first_q = collections.Counter()
    alive_end = collections.Counter()
    turn_uniq = []
    errs = 0
    guess_in_alive = 0
    guess_parsed = 0
    n_ok = 0
    out_tok = []
    in_tok = []
    for (k, s), r in eps.items():
        if r.get("error"):
            errs += 1
            print("ERROR", k, s, r["error"][:200])
            continue
        n_ok += 1
        rows = r["rows"]
        prev = None
        legal = [x for x in rows if not x.get("illegal")]
        if legal: first_q[legal[0]["query"]] += 1
        t_u = None
        for x in legal:
            if prev is not None:
                n_q += 1
                if x["query"] == prev + 1: n_scan += 1
            prev = x["query"]
            nb = x["n_before"]
            b = "1" if nb == 1 else "2" if nb == 2 else "3-5" if nb <= 5 else "6-15" if nb <= 15 else "16-100" if nb <= 100 else ">100"
            q_by_alive[b] += 1
            if x["zero_info"]: zero_by_alive[b] += 1
            if x["n_after"] == 1 and t_u is None: t_u = x["turn"]
        turn_uniq.append(t_u)
        res = r["result"]
        alive_end[res["n_alive_end"]] += 1
        out_tok.append(r["out_tokens"]); in_tok.append(r["in_tokens"])
        if res["parsed_guess"]: guess_parsed += 1
    print(f"== {name}: n_ok={n_ok} errs={errs}")
    print(f"  scan-next fraction (q == prev+1): {n_scan}/{n_q} = {n_scan/max(1,n_q):.3f}")
    print(f"  first query dist: {dict(first_q.most_common(5))}")
    print(f"  alive_end dist: {sorted(alive_end.items())}")
    tu = [t for t in turn_uniq if t is not None]
    print(f"  reached uniq: {len(tu)}; mean turn reached: {np.mean(tu) if tu else float('nan'):.1f}")
    for b in ["1","2","3-5","6-15","16-100",">100"]:
        print(f"   alive_before {b:>7}: queries={q_by_alive[b]:4d} zero={zero_by_alive[b]:4d} ({zero_by_alive[b]/max(1,q_by_alive[b]):.2f})")
    print(f"  out_tok mean {np.mean(out_tok):.0f}, in_tok mean {np.mean(in_tok):.0f}, max in {max(in_tok)}")
EOF
