import json, sys, collections, time
import numpy as np
sys.path.insert(0, "/Users/zarreboln/Desktop/ar_harness/stage0")
from cd_oracle import CDGame
insts = [json.loads(l) for l in open("data/cd_search.jsonl")]
games = {}
def game_for(comp):
    key = tuple(sorted(comp.items()))
    if key not in games: games[key] = CDGame(dict(key))
    return games[key]

def greedy_run(g, truth, first=None, budget=18):
    alive = np.arange(len(g.H))
    for t in range(1, budget+1):
        if len(alive) == 1: return t-1
        if t == 1 and first is not None: q = first
        else: _, q = g.available_bits(alive)
        alive = alive[g.H[alive, q] == truth[q]]
    return None if len(alive) > 1 else budget

res = collections.defaultdict(list)
for inst in insts:
    c = inst["controller"]
    g = game_for(c["components"])
    truth = np.array([int(ch) for ch in c["circuit_joint_lookup"]], dtype=np.uint8)
    n1 = greedy_run(g, truth)
    n2 = greedy_run(g, truth, first=0)
    res[tuple(sorted(c["components"].items()))].append((n1, n2))
for k, v in res.items():
    g = games[k]
    a = np.array(v, dtype=float)
    _, q0 = g.available_bits(np.arange(len(g.H)))
    print(k, "|H|=", len(g.H), "greedy first q:", q0, "n=", len(v),
          "greedy mean/max:", a[:,0].mean().round(2), a[:,0].max(), " A000-first mean/max:", a[:,1].mean().round(2), a[:,1].max())
    print("   dist A000-first:", sorted(collections.Counter(a[:,1].astype(int)).items()))
