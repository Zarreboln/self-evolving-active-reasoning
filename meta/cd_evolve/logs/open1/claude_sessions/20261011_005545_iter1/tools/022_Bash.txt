games = {}
def game_for(comp):
    key = tuple(sorted(comp.items()))
    if key not in games: games[key] = CDGame(dict(key))
    return games[key]
for name in sys.argv[1:]:
    eps = {}
    for line in open(f"logs/open1/traces/{name}/search.jsonl"):
        r = json.loads(line); eps[(r["key"], r["seed"])] = r
    stats = collections.Counter()
    for (k, s), r in eps.items():
        if r.get("error"): stats["error"] += 1; continue

        stats[f"{u}_guess_in_alive"] += in_alive
