import json, sys, collections
def load(p):
    latest={}
    for l in open(p):
        if l.strip():
            r=json.loads(l); latest[(r['key'],r['seed'])]=r
    return latest
for name in sys.argv[1:]:
    eps=load(name)
    n=len(eps)
    uniq=sum(bool((r.get('result') or {}).get('unique_at_end')) for r in eps.values())
    # zero-info by n_before bucket
    buck=collections.defaultdict(lambda:[0,0])
    seq=0; legal=0; zero=0; rep=0
    firstq=collections.Counter()
    turns_to_uniq=[]
    stall=[]
    for r in eps.values():
        rows=r.get('rows') or []
        prev=None
        for x in rows:
            if x.get('illegal'): continue
            if x['n_before']<=1: continue
            nb=x['n_before']
            b = '2' if nb==2 else '3-5' if nb<=5 else '6-15' if nb<=15 else '16-100' if nb<=100 else '>100'
            buck[b][0]+=1; buck[b][1]+=x['zero_info']
            legal+=1; zero+=x['zero_info']; rep+=x['repeat']
            if prev is not None and x['query']==prev+1: seq+=1
            prev=x['query']
        if rows and not rows[0].get('illegal'): firstq[rows[0]['query']]+=1
        # when did the set first reach 1
        t1=None
        for x in rows:
            if not x.get('illegal') and x['n_after']==1: t1=x['turn']; break
        turns_to_uniq.append(t1)
    print(name, 'n',n,'uniq',uniq/n, 'legal',legal,'zero',zero/legal,'rep',rep/legal,'seq(next idx)',seq/legal)
    for b in ['>100','16-100','6-15','3-5','2']:
        t,z=buck[b]; print('   n_before',b,'turns',t,'zero%',round(z/max(t,1),3))
    print('   first query', firstq.most_common(3))
    tt=[t for t in turns_to_uniq if t]; print('   reached uniq at turn: mean',sum(tt)/max(1,len(tt)), 'count',len(tt))
