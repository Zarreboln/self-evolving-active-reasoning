import json, sys
import numpy as np
sys.path.insert(0, '/Users/zarreboln/Desktop/ar_harness/stage0')
from cd_oracle import CDGame
insts=[json.loads(l) for l in open('/Users/zarreboln/Desktop/ar_harness/meta/cd_evolve/data/cd_search_v2.jsonl')]
games={}
def game(c):
    k=tuple(sorted(c.items()))
    if k not in games: games[k]=CDGame(dict(k))
    return games[k]
def det(H, alive, q):
    col=H[alive,q]; return col.min()==col.max()
def run(policy, rng, budget=18):
    res=[]
    for inst in insts:
        g=game(inst['controller']['components']); H=g.H
        truth=np.array([int(ch) for ch in inst['controller']['circuit_joint_lookup']],dtype=np.uint8)
        alive=np.arange(len(H)); hist=[]; ptr=0  # ptr = last proposed index
        state={'ptr':-1}
        for t in range(budget):
            if len(alive)==1: break
            q=policy(H, alive, hist, state, rng)
            hist.append(q)
            alive=alive[H[alive,q]==truth[q]]
        res.append(len(alive)==1)
    return np.mean(res)

def scan(H,alive,hist,st,rng):
    st['ptr']=(st['ptr']+1)%24; return st['ptr']
def scan_skip(H,alive,hist,st,rng):
    for _ in range(24):
        st['ptr']=(st['ptr']+1)%24
        if not det(H,alive,st['ptr']): return st['ptr']
    return st['ptr']
def make_veto(nveto, p_list, p_comply=1.0):
    # model proposes next in scan; if determined, vetoed: with prob p_comply it re-picks
    # (with prob p_list: first open entry in canonical order; else continues scan); else proposes something determined again
    def pol(H,alive,hist,st,rng):
        st['ptr']=(st['ptr']+1)%24; q=st['ptr']
        v=0
        while det(H,alive,q) and v<nveto:
            v+=1
            if rng.random()<p_comply:
                opens=[x for x in range(24) if not det(H,alive,x)]
                if rng.random()<p_list and opens:
                    q=opens[0]; st['ptr']=q
                else:
                    st['ptr']=(st['ptr']+1)%24; q=st['ptr']
            else:
                pass  # model insists / picks another determined: keep q determined
        return q
    return pol
def rand_open(H,alive,hist,st,rng):
    opens=[x for x in range(24) if not det(H,alive,x)]
    return int(rng.choice(opens)) if opens else 0
rng=np.random.default_rng(0)
print('scan', run(scan,rng))
print('scan_skip_det', run(scan_skip,rng))
print('rand_open', np.mean([run(rand_open,np.random.default_rng(s)) for s in range(3)]))
for nv in (1,2):
    for pl in (0.0,0.5,1.0):
        for pc in (1.0,0.8,0.6):
            print(f'veto n={nv} p_list={pl} p_comply={pc}', np.mean([run(make_veto(nv,pl,pc),np.random.default_rng(s)) for s in range(3)]))
