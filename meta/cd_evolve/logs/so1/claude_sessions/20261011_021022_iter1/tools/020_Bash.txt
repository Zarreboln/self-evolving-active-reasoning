import json, sys, collections, re
import numpy as np
def load(p):
    latest={}
    for l in open(p):
        if l.strip():
            r=json.loads(l); latest[(r['key'],r['seed'])]=r
    return latest
for name in sys.argv[1:]:
    eps=load(name)
    errs=[(k,r['error'][:120]) for k,r in eps.items() if r.get('error')]
    mcc=[(r.get('result') or {}).get('max_context_chars',0) for r in eps.values() if r.get('result')]
    reply_len=[]; guess_in_reply=0; nrep=0; sent_len=[]
    for r in eps.values():
        for x in r.get('rows') or []:
            reply_len.append(len(x.get('model_text') or ''))
            if '<guess>' in (x.get('model_text') or ''): guess_in_reply+=1
            if x.get('sent'): sent_len.append(len(x['sent']))
            nrep+=1
    print(name)
    print('  errors', len(errs), errs[:3])
    print('  max_context_chars: mean %.0f p90 %.0f max %.0f'%(np.mean(mcc),np.percentile(mcc,90),np.max(mcc)))
    print('  reply chars mean %.0f p90 %.0f'%(np.mean(reply_len),np.percentile(reply_len,90)), 'replies with <guess>: %.3f'%(guess_in_reply/nrep))
    print('  sent chars mean %.0f max %.0f'%(np.mean(sent_len),np.max(sent_len)))
    print('  tok_in mean %.0f tok_out mean %.0f'%(np.mean([r.get('in_tokens',0) for r in eps.values()]),np.mean([r.get('out_tokens',0) for r in eps.values()])))
    # inventory breakdown of uniq
    inv=collections.defaultdict(list)
    for r in eps.values():
        res=r.get('result') or {}
        # infer inventory from first row? use n_before of turn1
        rows=r.get('rows') or []
        nb=rows[0]['n_before'] if rows and 'n_before' in rows[0] else None
        inv[nb].append(bool(res.get('unique_at_end')))
    print('  uniq by |H|', {k:(len(v),round(np.mean(v),3)) for k,v in inv.items()})
