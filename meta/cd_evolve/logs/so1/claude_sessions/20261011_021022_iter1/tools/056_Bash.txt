import json, sys, numpy as np
SYS_CHARS = 2600   # filled system prompt, approx
TPC_IN = 0.45      # tokens per char, mixed context (memory: 0.55 harness text, 0.40 replies/env)
def load(p):
    latest={}
    for l in open(p):
        if l.strip():
            r=json.loads(l); latest[(r['key'],r['seed'])]=r
    return latest
for name, base_cost, r_per_zero in [('enum15', 65946, 1.3), ('bare', 34302, 1.3), ('bare', 34302, 1.6)]:
    eps=load(f'/Users/zarreboln/Desktop/ar_harness/meta/cd_evolve/logs/so1/traces/{name}/search.jsonl')
    extra=[]; nre=[]
    for r in eps.values():
        ctx = SYS_CHARS + 40
        e = 0.0; n = 0
        for x in r.get('rows') or []:
            reply = len(x.get('model_text') or '')
            if not x.get('illegal') and x.get('zero_info') and x['n_before'] > 1:
                # one re-ask = context + rejected reply + veto msg (~450 chars) in, ~reply/2 out
                e += r_per_zero * ((ctx + reply + 450) * TPC_IN + 0.3 * reply)
                n += r_per_zero
            ctx += reply + len(x.get('sent') or '')
        extra.append(e); nre.append(n)
    print(f"{name} r/zero={r_per_zero}: re-asks/ep {np.mean(nre):.1f}  extra tokens/ep {np.mean(extra):.0f}  -> cost {base_cost+np.mean(extra):.0f} ({(base_cost+np.mean(extra))/65946-1:+.0%} vs enum15)")
