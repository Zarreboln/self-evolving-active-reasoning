from __future__ import annotations

from abc import ABC, abstractmethod


class BudgetExhausted(Exception):
    pass


class CDHarness(ABC):
    def __init__(self, env, solver):
        self.env = env
        self._solver = solver          # callable(messages, temperature, max_tokens) -> str
        self._trace = []

    def llm(self, messages, temperature=0.7, max_tokens=1200):
        snapshot = [dict(m) for m in messages]      # the harness keeps appending to its list
        out = self._solver(snapshot, temperature=temperature, max_tokens=max_tokens)
        self._trace.append({"step": "llm", "messages": snapshot, "response": out})
        return out

    @abstractmethod
    def solve(self) -> str:
        """Play the episode through self.env / self.llm and return the final guess."""

class Harness(CDHarness):
    def solve(self):
        env = self.env
        msgs = [{"role": "system", "content": env.system_prompt},
                {"role": "user", "content": "Please start with your first query."}]
        last_line = None
        B = env.budget
        for t in range(1, B + 1):
            reply = self.llm(msgs)
            msgs.append({"role": "assistant", "content": reply})
            m = QUERY_RE.search(reply or "")
            if not m:
                # the benchmark gives no second chance: the turn is spent
                msgs.append({"role": "user", "content": "Your message did not contain a query in the "
                             "required format. Reply with exactly one <query>X(a, b, c)</query>."})
                if t == B:
                    last_line = None      # the previous answer was already delivered with its own prompt
                continue

            try:
                bit = env.query(c, a, b, x)
            except Exception:
                break

                msgs.append({"role": "user", "content": f"{last_line}\nPlease send your next query. {B - t} queries remaining."})

        msgs.append({"role": "user", "content": final})
